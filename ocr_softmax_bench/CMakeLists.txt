cmake_minimum_required(VERSION 3.16)
project(ocr_softmax_bench LANGUAGES CXX)

set(OCR_SOFTMAX_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
set(OPENCL_HEADERS_DIR "${OCR_SOFTMAX_ROOT}/third_party/OpenCL-Headers" CACHE PATH "Khronos OpenCL-Headers")

add_executable(ocr_softmax_bench softmax_bench.cpp)
target_compile_features(ocr_softmax_bench PRIVATE cxx_std_17)

if(EXISTS "${OPENCL_HEADERS_DIR}/CL/cl.h")
    add_executable(ocl_test_2d device_test/ocl_test_2d.cpp)
    target_compile_features(ocl_test_2d PRIVATE cxx_std_17)
    target_include_directories(ocl_test_2d PRIVATE "${OPENCL_HEADERS_DIR}")
    target_compile_definitions(ocl_test_2d PRIVATE CL_TARGET_OPENCL_VERSION=120)

    if(OHOS OR CMAKE_SYSTEM_NAME STREQUAL "OHOS")
        target_sources(ocl_test_2d PRIVATE device_test/opencl_dynload.cpp)
        target_compile_definitions(ocl_test_2d PRIVATE OCR_OPENCL_DLOPEN=1)
        target_link_libraries(ocl_test_2d PRIVATE dl)
        message(STATUS "ocl_test_2d: OHOS dynamic OpenCL (dlopen libOpenCL.so)")
    elseif(DEFINED OPENCL_LIBRARY AND OPENCL_LIBRARY)
        target_link_libraries(ocl_test_2d PRIVATE "${OPENCL_LIBRARY}")
        message(STATUS "ocl_test_2d: link ${OPENCL_LIBRARY}")
    else()
        target_link_libraries(ocl_test_2d PRIVATE OpenCL)
        message(STATUS "ocl_test_2d: link -lOpenCL")
    endif()
else()
    message(WARNING "OpenCL headers missing — ocl_test_2d not built. Clone third_party/OpenCL-Headers")
endif()
