cmake_minimum_required(VERSION 3.16)
project(ocr_softmax_bench LANGUAGES CXX)

set(OCR_SOFTMAX_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
set(OPENCL_HEADERS_DIR "${OCR_SOFTMAX_ROOT}/third_party/OpenCL-Headers" CACHE PATH "Khronos OpenCL-Headers")

add_executable(ocr_softmax_bench softmax_bench.cpp)
target_compile_features(ocr_softmax_bench PRIVATE cxx_std_17)

if(EXISTS "${OPENCL_HEADERS_DIR}/CL/cl.h")
    function(configure_ocr_ocl_target target_name)
        add_executable(${target_name} device_test/${target_name}.cpp)
        target_compile_features(${target_name} PRIVATE cxx_std_17)
        target_include_directories(${target_name} PRIVATE "${OPENCL_HEADERS_DIR}" "${OCR_SOFTMAX_ROOT}/device_test")
        target_compile_definitions(${target_name} PRIVATE CL_TARGET_OPENCL_VERSION=120)

        if(OHOS OR CMAKE_SYSTEM_NAME STREQUAL "OHOS")
            target_sources(${target_name} PRIVATE device_test/opencl_dynload.cpp)
            target_compile_definitions(${target_name} PRIVATE OCR_OPENCL_DLOPEN=1)
            target_link_libraries(${target_name} PRIVATE dl)
            message(STATUS "${target_name}: OHOS dynamic OpenCL (dlopen libOpenCL.so)")
        elseif(DEFINED OPENCL_LIBRARY AND OPENCL_LIBRARY)
            target_link_libraries(${target_name} PRIVATE "${OPENCL_LIBRARY}")
            message(STATUS "${target_name}: link ${OPENCL_LIBRARY}")
        else()
            target_link_libraries(${target_name} PRIVATE OpenCL)
            message(STATUS "${target_name}: link -lOpenCL")
        endif()
    endfunction()

    configure_ocr_ocl_target(ocl_test_2d)
    configure_ocr_ocl_target(ocl_test_ctc)
else()
    message(WARNING "OpenCL headers missing at ${OPENCL_HEADERS_DIR}/CL/cl.h — ocl_test_2d not built")
endif()
