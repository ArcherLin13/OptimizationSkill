cmake_minimum_required(VERSION 3.16)
project(findmax_bench LANGUAGES CXX)

set(ROOT ${CMAKE_CURRENT_SOURCE_DIR})
set(OPENCL_HEADERS_DIR "${ROOT}/../ocr_softmax_bench/third_party/OpenCL-Headers" CACHE PATH
    "Khronos OpenCL-Headers")

if(NOT EXISTS "${OPENCL_HEADERS_DIR}/CL/cl.h")
    message(FATAL_ERROR "OpenCL headers missing: ${OPENCL_HEADERS_DIR}/CL/cl.h")
endif()

function(add_findmax_ocl_test target_name src_file)
    add_executable(${target_name} ${src_file})
    target_compile_features(${target_name} PRIVATE cxx_std_17)
    target_include_directories(${target_name} PRIVATE "${OPENCL_HEADERS_DIR}" "${ROOT}/device_test")
    target_compile_definitions(${target_name} PRIVATE CL_TARGET_OPENCL_VERSION=120)
    if(OHOS OR CMAKE_SYSTEM_NAME STREQUAL "OHOS")
        target_sources(${target_name} PRIVATE device_test/opencl_dynload.cpp)
        target_compile_definitions(${target_name} PRIVATE OCR_OPENCL_DLOPEN=1)
        target_link_libraries(${target_name} PRIVATE dl)
        message(STATUS "${target_name}: OHOS dynamic OpenCL (dlopen)")
    elseif(DEFINED OPENCL_LIBRARY AND OPENCL_LIBRARY)
        target_link_libraries(${target_name} PRIVATE "${OPENCL_LIBRARY}")
    else()
        target_link_libraries(${target_name} PRIVATE OpenCL)
    endif()
endfunction()

add_findmax_ocl_test(ocl_test_findmax device_test/ocl_test_findmax.cpp)
add_findmax_ocl_test(ocl_test_enhance device_test/ocl_test_enhance.cpp)
